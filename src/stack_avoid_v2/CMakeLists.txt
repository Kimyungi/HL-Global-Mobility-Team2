cmake_minimum_required(VERSION 3.16)
project(stack_avoid_v2 LANGUAGES CXX)
# The planner has a 20ms execution budget; default colcon builds need optimization.
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
option(AVOID_V2_CORE_ONLY "Build portable planner and tests without ROS" OFF)
add_library(avoid_v2_core src/core.cpp)
target_include_directories(avoid_v2_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
enable_testing()
add_executable(avoid_v2_return_tracker_test test/return_tracker_test.cpp)
target_link_libraries(avoid_v2_return_tracker_test avoid_v2_core)
add_test(NAME avoid_v2_return_tracker_test COMMAND avoid_v2_return_tracker_test)
add_executable(avoid_v2_core_test test/core_test.cpp)
target_link_libraries(avoid_v2_core_test avoid_v2_core)
add_test(NAME avoid_v2_core_test COMMAND avoid_v2_core_test)
add_executable(avoid_v2_scenarios test/scenarios.cpp)
target_link_libraries(avoid_v2_scenarios avoid_v2_core)
add_test(NAME avoid_v2_scenarios COMMAND avoid_v2_scenarios)
add_executable(avoid_v2_zone_scenarios test/zone_scenarios.cpp)
target_link_libraries(avoid_v2_zone_scenarios avoid_v2_core)
add_test(NAME avoid_v2_zone_scenarios COMMAND avoid_v2_zone_scenarios)
add_executable(avoid_v2_s_lateral_scenarios test/s_lateral_scenarios.cpp)
target_link_libraries(avoid_v2_s_lateral_scenarios avoid_v2_core)
add_test(NAME avoid_v2_s_lateral_scenarios COMMAND avoid_v2_s_lateral_scenarios)
add_executable(avoid_v2_same_vehicle_scenarios test/same_vehicle_scenarios.cpp)
target_link_libraries(avoid_v2_same_vehicle_scenarios avoid_v2_core)
add_test(NAME avoid_v2_same_vehicle_scenarios COMMAND avoid_v2_same_vehicle_scenarios)
add_executable(avoid_v2_wall_corridor_test test/wall_corridor_test.cpp)
target_link_libraries(avoid_v2_wall_corridor_test avoid_v2_core)
add_test(NAME avoid_v2_wall_corridor_test COMMAND avoid_v2_wall_corridor_test)
add_executable(avoid_v2_adaptive_speed_test test/adaptive_speed_test.cpp)
target_link_libraries(avoid_v2_adaptive_speed_test avoid_v2_core)
add_test(NAME avoid_v2_adaptive_speed_test COMMAND avoid_v2_adaptive_speed_test)
add_executable(avoid_v2_execution_safety_test test/execution_safety_test.cpp)
target_link_libraries(avoid_v2_execution_safety_test avoid_v2_core)
add_test(NAME avoid_v2_execution_safety_test COMMAND avoid_v2_execution_safety_test)
add_executable(avoid_v2_export tools/export_lab.cpp)
target_link_libraries(avoid_v2_export avoid_v2_core)
add_executable(avoid_v2_probe_s_lateral tools/probe_s_lateral.cpp)
target_link_libraries(avoid_v2_probe_s_lateral avoid_v2_core)
if(NOT AVOID_V2_CORE_ONLY)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(fma_interfaces REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(rcl_interfaces REQUIRED)
  find_package(std_msgs REQUIRED)
  add_executable(avoid_v2_node src/node.cpp)
  target_link_libraries(avoid_v2_node avoid_v2_core)
  ament_target_dependencies(avoid_v2_node rclcpp fma_interfaces sensor_msgs nav_msgs geometry_msgs rcl_interfaces std_msgs)
  install(TARGETS avoid_v2_node DESTINATION lib/${PROJECT_NAME})
  install(PROGRAMS tools/publish_course.py tools/watchdog.py DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
  install(DIRECTORY include/ DESTINATION include)
  install(TARGETS avoid_v2_core EXPORT avoid_v2_targets ARCHIVE DESTINATION lib)
  ament_export_targets(avoid_v2_targets HAS_LIBRARY_TARGET)
  ament_package()
endif()
